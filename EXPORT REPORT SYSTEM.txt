STEP 1: CREATE lib/reportUtils.ts
─────────────────────────────────────────────────────────────

Export these functions:

// Filter items for a specific week (Mon–Sun)
getWeeklyItems(
  items: ContentItem[], 
  weekStart: Date
): ContentItem[]

// Filter items for a specific month
getMonthlyItems(
  items: ContentItem[], 
  year: number, 
  month: number
): ContentItem[]

// Aggregate stats from a list of items
aggregateStats(items: ContentItem[]): {
  total: number
  published: number
  scheduled: number
  pendingApproval: number
  inProduction: number
  needsRework: number
  planned: number
  kpiPending: number
  byPillar: Record<ContentPillar, number>
  byFormat: Record<ContentFormat, number>
  byStatus: Record<ContentStatus, number>
  byPlatform: Record<string, number>
}

// Calculate performance averages from items with performance data
aggregatePerformance(items: ContentItem[]): {
  avgEngagementRate: number
  avgSaveRate: number
  avgShareRate: number
  avgWatchTime: number
  totalReach: number
  totalSaves: number
  kpiPassRate: number           // % of KPIs passed overall
  topPost: ContentItem | null   // highest engagement rate
  worstPost: ContentItem | null // lowest engagement rate
  byPillar: Record<ContentPillar, {
    avgEngagementRate: number
    kpiPassRate: number
    count: number
  }>
  byFormat: Record<ContentFormat, {
    avgEngagementRate: number
    count: number
  }>
}

// Get at-risk items (deadline within 2 days or overdue)
getAtRiskItems(items: ContentItem[]): {
  overdue: ContentItem[]   // deadline passed, not published
  critical: ContentItem[]  // deadline within 1 day
  warning: ContentItem[]   // deadline within 3 days
}

// Get top N posts by engagement rate
getTopPosts(items: ContentItem[], n: number): ContentItem[]

// Generate action recommendations based on data
generateRecommendations(
  stats: ReturnType<typeof aggregateStats>,
  perf: ReturnType<typeof aggregatePerformance>
): string[]
// Rules for recommendations:
// - If any pillar has 0 content → "Add content for [Pillar]"
// - If kpiPassRate < 50% → "Review content strategy for low KPIs"
// - If needsRework > 3 → "High revision rate — review Brief quality"
// - If pendingApproval > 5 → "Approval bottleneck detected"
// - Top performing pillar → "Double down on [Pillar] content"

STEP 17: INJECT PRINT STYLES ON EXPORT

In ExportModal.tsx, implement the handleExport function:

const handleExport = () => {
  // 1. Inject print styles
  const styleEl = document.createElement('div')
  styleEl.innerHTML = PRINT_STYLES
  document.head.appendChild(styleEl.firstChild as HTMLElement)

  // 2. Add print root ID to the report container
  const reportEl = document.getElementById('report-print-root')

  // 3. Set correct page orientation via @page rule
  const orientationStyle = document.createElement('style')
  orientationStyle.id = 'report-orientation'
  orientationStyle.innerHTML = reportType === 'monthly'
    ? `@page { size: A4 portrait; }`
    : `@page { size: A4 portrait; }`
  document.head.appendChild(orientationStyle)

  // 4. Small delay to ensure styles are applied before print dialog
  setTimeout(() => {
    window.print()

    // 5. Cleanup after print dialog closes
    const cleanup = () => {
      document.getElementById('report-print-styles')?.remove()
      document.getElementById('report-orientation')?.remove()
      window.removeEventListener('afterprint', cleanup)
    }
    window.addEventListener('afterprint', cleanup)
  }, 300)
}
